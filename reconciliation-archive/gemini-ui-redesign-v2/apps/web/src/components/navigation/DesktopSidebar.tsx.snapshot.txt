'use client';

import React from 'react';
import Link from 'next/link';
import { usePathname } from 'next/navigation';
import {
  Home,
  BookOpen,
  RotateCcw,
  Brain,
  TrendingUp,
  Library,
  Search,
  Settings,
  User,
  Sparkles,
} from 'lucide-react';
import { primaryNavItems, secondaryNavItems } from './NavTypes';

const iconMap: Record<string, React.ComponentType<{ className?: string }>> = {
  Home,
  BookOpen,
  RotateCcw,
  Brain,
  TrendingUp,
  Library,
  Search,
  Settings,
  User,
};

export interface DesktopSidebarProps {
  currentPath?: string;
}

export function DesktopSidebar({ currentPath }: DesktopSidebarProps = {}) {
  let pathname = usePathname();
  if (!pathname && currentPath) {
    pathname = currentPath;
  }

  return (
    <aside
      aria-label="Điều hướng màn hình lớn"
      className="hidden xl:flex flex-col w-64 fixed left-0 top-0 bottom-0 z-30 bg-[var(--surface)] border-r border-[var(--border-subtle)] p-5 justify-between backdrop-blur-md"
    >
      <div className="flex flex-col gap-5">
        {/* Brand Header */}
        <Link
          href="/today"
          className="flex items-center gap-3 px-2 py-1.5 rounded-2xl transition-all hover:bg-[var(--surface-subtle)] focus-visible:outline-none focus-visible:ring-2 focus-visible:ring-[var(--focus-ring)] group"
        >
          <div className="w-10 h-10 rounded-2xl bg-gradient-to-br from-indigo-500 to-indigo-700 flex items-center justify-center text-white font-bold text-lg shadow-sm shadow-indigo-500/20 ring-1 ring-white/20 group-hover:scale-[1.02] transition-transform">
            L
          </div>
          <div className="flex flex-col">
            <div className="flex items-center gap-1.5">
              <h1 className="font-bold text-base tracking-tight leading-none text-[var(--text-primary)]">
                LINGUA OS
              </h1>
              <span className="text-[10px] font-semibold font-mono uppercase px-1.5 py-0.5 rounded-md bg-[var(--status-info-bg)] text-[var(--action-primary)] border border-[var(--status-info-border)]">
                FSRS
              </span>
            </div>
            <p className="text-[11px] text-[var(--text-secondary)] mt-1">Adaptive Workspace</p>
          </div>
        </Link>

        {/* Quick Command / Search Trigger */}
        <Link
          href="/dictionary"
          className="flex items-center justify-between px-3 py-2 text-xs text-[var(--text-secondary)] bg-[var(--surface-subtle)] rounded-xl border border-[var(--border-subtle)] hover:border-[var(--border-default)] hover:text-[var(--text-primary)] transition-colors focus-visible:outline-none focus-visible:ring-2 focus-visible:ring-[var(--focus-ring)]"
        >
          <span className="flex items-center gap-2">
            <Search className="w-3.5 h-3.5 text-[var(--text-muted)]" />
            <span>Tra cứu từ vựng...</span>
          </span>
          <kbd className="px-1.5 py-0.5 text-[10px] font-mono bg-[var(--surface)] text-[var(--text-muted)] rounded border border-[var(--border-subtle)] shadow-xs">
            ⌘K
          </kbd>
        </Link>

        {/* Primary Navigation */}
        <nav aria-label="Menu chính" className="flex flex-col gap-1">
          <span className="text-[10px] font-bold uppercase tracking-wider text-[var(--text-muted)] px-3 mb-1">
            Không gian học tập
          </span>
          {primaryNavItems.map((item) => {
            const Icon = iconMap[item.iconName] || Home;
            const isActive = pathname === item.href || (item.href !== '/' && pathname?.startsWith(`${item.href}`));

            return (
              <Link
                key={item.href}
                href={item.href}
                aria-current={isActive ? 'page' : undefined}
                className={`relative flex items-center gap-3 px-3 py-2.5 rounded-xl text-sm font-medium transition-all focus-visible:outline-none focus-visible:ring-2 focus-visible:ring-[var(--focus-ring)] ${
                  isActive
                    ? 'bg-[var(--status-info-bg)] text-[var(--action-primary)] font-semibold shadow-xs border border-[var(--status-info-border)]/60'
                    : 'text-[var(--text-secondary)] hover:bg-[var(--surface-subtle)] hover:text-[var(--text-primary)]'
                }`}
              >
                {isActive && (
                  <span className="absolute left-0 top-2.5 bottom-2.5 w-1 rounded-r-full bg-[var(--action-primary)]" />
                )}
                <Icon className={`w-5 h-5 ${isActive ? 'stroke-[2.25px] text-[var(--action-primary)]' : 'stroke-[1.75px] text-[var(--text-muted)]'}`} />
                <span>{item.labelVi}</span>
              </Link>
            );
          })}
        </nav>

        {/* Secondary Navigation */}
        <nav aria-label="Khám phá và cài đặt" className="flex flex-col gap-1 pt-3 border-t border-[var(--border-subtle)]">
          <span className="text-[10px] font-bold uppercase tracking-wider text-[var(--text-muted)] px-3 mb-1">
            Tài liệu & Quản trị
          </span>
          {secondaryNavItems.map((item) => {
            const Icon = iconMap[item.iconName] || User;
            const isActive = pathname === item.href || (item.href !== '/' && pathname?.startsWith(`${item.href}`));

            return (
              <Link
                key={item.href}
                href={item.href}
                aria-current={isActive ? 'page' : undefined}
                className={`flex items-center gap-3 px-3 py-2 rounded-xl text-sm font-medium transition-colors focus-visible:outline-none focus-visible:ring-2 focus-visible:ring-[var(--focus-ring)] ${
                  isActive
                    ? 'bg-[var(--surface-subtle)] text-[var(--text-primary)] font-semibold'
                    : 'text-[var(--text-secondary)] hover:bg-[var(--surface-subtle)] hover:text-[var(--text-primary)]'
                }`}
              >
                <Icon className="w-4 h-4 text-[var(--text-muted)]" />
                <span>{item.labelVi}</span>
              </Link>
            );
          })}
        </nav>
      </div>

      {/* Footer Profile / Brain Summary Card */}
      <Link
        href="/profile"
        aria-label="Xem hồ sơ người học"
        className="p-3.5 rounded-2xl bg-[var(--surface-subtle)] border border-[var(--border-subtle)] flex items-center justify-between hover:border-[var(--border-strong)] transition-all focus-visible:outline-none focus-visible:ring-2 focus-visible:ring-[var(--focus-ring)] group shadow-xs hover:shadow"
      >
        <div className="flex items-center gap-2.5">
          <div className="w-9 h-9 rounded-xl bg-gradient-to-br from-indigo-50 to-sky-100 text-[var(--action-primary)] flex items-center justify-center font-bold text-xs border border-[var(--status-info-border)] shadow-xs">
            A2
          </div>
          <div>
            <div className="text-xs font-semibold text-[var(--text-primary)] truncate max-w-[120px]">
              Learner Profile
            </div>
            <div className="text-[11px] text-[var(--text-secondary)] flex items-center gap-1 mt-0.5">
              <Sparkles className="w-3 h-3 text-[var(--status-warning)]" />
              <span>483 từ đã củng cố</span>
            </div>
          </div>
        </div>
        <div className="p-1.5 rounded-lg text-[var(--text-secondary)] group-hover:bg-[var(--surface)] group-hover:text-[var(--text-primary)] transition-colors">
          <User className="w-4 h-4" />
        </div>
      </Link>
    </aside>
  );
}
